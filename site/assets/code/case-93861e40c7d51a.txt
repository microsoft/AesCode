    function gauge(id, value, color) {
      var chart = echarts.init(document.getElementById(id), null, { renderer: 'canvas' });
      chart.setOption({
        animation: false,
        series: [{
          type: 'pie',
          radius: ['52%', '74%'],
          center: ['50%', '82%'],
          startAngle: 180,
          clockwise: true,
          avoidLabelOverlap: false,
          label: { show: false },
          labelLine: { show: false },
          data: [value, 100 - value],
          itemStyle: {
            color: function(params) {
              return params.dataIndex === 0 ? color : '#edf0f5';
            },
            borderWidth: 2,
            borderColor: '#ffffff'
          }
        }]
      });
    }

    gauge('gauge1', 82, '#3170ff');
    gauge('gauge2', 70, '#3170ff');
    gauge('gauge3', 93.2, '#39c766');
